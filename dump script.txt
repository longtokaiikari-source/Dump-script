-- Dragging
local dragInput, dragStart, startPos
local function update(input)
	local delta = input.Position - dragStart
	local targetPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	TweenService:Create(MainFrame, TweenInfo.new(0.08, Enum.EasingStyle.Quad), { Position = targetPos }):Play()
end
TitleBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragStart = input.Position
		startPos = MainFrame.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then dragStart = nil end
		end)
	end
end)
TitleBar.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		dragInput = input
	end
end)
game:GetService("UserInputService").InputChanged:Connect(function(input)
	if input == dragInput and dragStart then update(input) end
end)

-- Tab switching
local activeTab = "Dashboard"
local tabButtons = {}

local function switchTab(tabName)
	activeTab = tabName
	DashboardTab.Visible = (tabName == "Dashboard")
	SettingsTab.Visible  = (tabName == "Settings")
	FiltersTab.Visible   = (tabName == "Filters")
	LogsTab.Visible      = (tabName == "Logs")
	for name, btn in pairs(tabButtons) do
		if name == tabName then
			TweenService:Create(btn, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
				BackgroundColor3 = Color3.fromRGB(40, 30, 60),
				TextColor3 = Color3.fromRGB(200, 120, 255)
			}):Play()
			btn.UIStroke.Enabled = true
		else
			TweenService:Create(btn, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
				BackgroundColor3 = Color3.fromRGB(22, 22, 30),
				TextColor3 = Color3.fromRGB(160, 160, 180)
			}):Play()
			btn.UIStroke.Enabled = false
		end
	end
end

local function createTabButton(name, text)
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(1, 0, 0, 36)
	btn.BackgroundColor3 = Color3.fromRGB(22, 22, 30)
	btn.Text = text

	btn.Parent = Sidebar
	Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)
	local stroke = Instance.new("UIStroke")
	stroke.Name = "UIStroke"
	stroke.Color = Color3.fromRGB(150, 80, 255)
	stroke.Thickness = 1
	stroke.Enabled = false
	stroke.Parent = btn
	btn.MouseEnter:Connect(function()
		if activeTab ~= name then
			TweenService:Create(btn, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
				BackgroundColor3 = Color3.fromRGB(28, 28, 38),
				TextColor3 = Color3.fromRGB(180, 180, 200)
			}):Play()
		end
	end)
	btn.MouseLeave:Connect(function()
		if activeTab ~= name then
			TweenService:Create(btn, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
				BackgroundColor3 = Color3.fromRGB(22, 22, 30),
				TextColor3 = Color3.fromRGB(160, 160, 180)
			}):Play()
		end
	end)
	btn.MouseButton1Click:Connect(function() switchTab(name) end)
	tabButtons[name] = btn
end

local setPhaseStatus, markPhaseActive, markPhaseDone, resetPhaseUI

setPhaseStatus = function(key, countOrText, total)
	local row = phaseRows[key]
	if not row then return end
	if total then
		row.statusLbl.Text = tostring(countOrText) .. "/" .. tostring(total)
	else
		row.statusLbl.Text = tostring(countOrText)
	end
end

markPhaseActive = function(key)

	if not row then return end
	TweenService:Create(row.row,    TweenInfo.new(0.3), { BackgroundColor3 = Color3.fromRGB(28, 20, 45) }):Play()

local function ensureDir(path)
	if not isfolder(path) then
		pcall(makefolder, path)
	end
end

local function createFoldersForPath(fullPath)
	local parts = string.split(fullPath, "/")
	local current = parts[1]
	for i = 2, #parts - 1 do
		current = current .. "/" .. parts[i]
		ensureDir(current)
	end
end

local function getUniquePath(path, ext)
	if not isfile(path .. ext) then return path .. ext end
	local counter = 1
	while isfile(path .. "_" .. tostring(counter) .. ext) do
		counter += 1
	end
	return path .. "_" .. tostring(counter) .. ext
end

local function safeStr(s)
	s = tostring(s or "unknown")
	s = string.gsub(s, "[%c]", "")
	s = string.gsub(s, "[^%w%.%-%_]", "_")
	s = string.gsub(s, "^_+", "")
	if s == "" or s == "_" or s == ".lua" or s == ".luac" or s == ".luau" then
		s = "Script_" .. tostring(math.random(100000, 999999))
	end
	if #s > 100 then s = string.sub(s, 1, 100) end
	return s
end

local function fullNameOf(inst)
	local ok, fn = pcall(function() return inst:GetFullName() end)
	if ok and fn then return fn end
	return "<unnamed>"
end

local function classOf(inst)
	local ok, cn = pcall(function() return inst.ClassName end)
	if ok and cn then return cn end
	return "?"
end

-- appendfile is one write per line; the read-whole-file-then-rewrite path stays as fallback
local function logToFile(fileName, content)
	local appended = false
	if appendfile then
		appended = pcall(appendfile, fileName, content .. "\n\n")
	end
	if not appended then
		local current = ""
		pcall(function()
			if isfile(fileName) then current = readfile(fileName) or "" end
		end)
		pcall(writefile, fileName, current .. content .. "\n\n")
	end
end

local function bcHash(bc)
	if not bc or #bc == 0 then return nil end
	return tostring(#bc) .. "_" .. string.sub(bc, 1, 64)
end

local function cleanSource(src)
	if not Config.CleanHeaders then return src end
	src = src:gsub("^%-%- Decompiled with[%s%S]-%\n%\n", "")
	src = src:gsub("^%-%- Decompiled by[%s%S]-%\n%\n",   "")
	src = src:gsub("^%-%- Decompiler[%s%S]-%\n%\n",      "")
	return src
end

local function isUselessSrc(src)
	if type(src) ~= "string" or #src == 0 then return true end
	local stripped = string.gsub(src, "%s+", "")
	if #stripped < 8 then return true end
	if #stripped < 200 then
		local lower = string.lower(stripped)
		if string.find(lower, "failed", 1, true)
			or string.find(lower, "unable", 1, true)
			or string.find(lower, "error", 1, true)
			or string.find(lower, "cannotdecompile", 1, true) then
			return true
		end
	end
	return false
end

local function unwrapProxy(obj)
	if obj == nil then return nil end
	local realObj = obj
	local ok, mt = pcall(getrawmetatable, obj)
	if ok and mt then
		local ok2, real = pcall(function()
			local indexFn = rawget(mt, "__index")
			if type(indexFn) == "function" then
				for i = 1, 30 do
					local n, v = debug.getupvalue(indexFn, i)
					if n == nil then break end
					if type(v) == "table" and v.GetObject then
						return v:GetObject()
					end
				end
			end
		end)
		if ok2 and real and real ~= obj then realObj = real end
	end
	return realObj
end

local lastYield = os.clock()

local function checkYield()
	local sp = ThreadSpeeds[Config.ThreadingSpeed] or ThreadSpeeds.Medium
	if os.clock() - lastYield > 0.016 then
		task.wait(sp.YieldTime)
		lastYield = os.clock()
	end
end

local ThreatKeywords = {
	{ pattern = "loadstring",                   name = "Dynamic Execution (loadstring)" },
	{ pattern = "getfenv",                       name = "Environment Access (getfenv)" },
	{ pattern = "setfenv",                       name = "Environment Manipulation (setfenv)" },
	{ pattern = "HttpService",                   name = "External Communication (HttpService)" },
	{ pattern = "PostAsync",                     name = "Data Sending (PostAsync)" },
	{ pattern = "GetAsync",                      name = "Data Fetching (GetAsync)" },
	{ pattern = "DiscordApp",                    name = "Possible Webhook Reference" },
	{ pattern = "MarketplaceService",            name = "Monetization Reference" },
	{ pattern = "require%s*%(%s*%d+",            name = "Obfuscated/External Module Require" },
}

local function getLinePreview(src, pattern)
	local ok, lines = pcall(string.split, src, "\n")
	if not ok or not lines then return "Match found" end
	for i, l in ipairs(lines) do
		if string.find(l, pattern) then
			return "Line " .. tostring(i) .. ": " .. string.sub(string.gsub(l, "^%s+", ""), 1, 120)
		end
	end
	return "Match found"
end

-- Get full 1:1 game path from instance using GetFullName()
local function getFullGamePath(inst)
	local ok, fn = pcall(function() return inst:GetFullName() end)
	if not ok or not fn or fn == "" then return "" end
	-- Remove "game." prefix → "ReplicatedStorage.Folder.Script"
	fn = fn:gsub("^game%.", ""):gsub("^game$", "")
	-- Split by dots and sanitize each segment
	local parts = string.split(fn, ".")
	local safeParts = {}
	for _, p in ipairs(parts) do
		local sp = safeStr(p)
		if sp ~= "" then table.insert(safeParts, sp) end
	end
	if #safeParts == 0 then return "" end
	return table.concat(safeParts, "/")
end

-- =================================================================
-- PHASE 1: SCAN - Collect all unique scripts
-- =================================================================
local function runScanPhase(queue)
	local seenPtr  = {}     -- [instance] = true (pointer dedup)
	local scanTotal = 0

	local function isScriptType(inst)
		local ok, r = pcall(function()
			return inst:IsA("LocalScript") or inst:IsA("ModuleScript") or inst:IsA("Script")
		end)
		return ok and r
	end

	local function tryAddToQueue(inst, tag)
		if seenPtr[inst] then return end

		-- Get bytecode for hash deduplication
		local bc = ""
		if getscriptbytecode then
			local realI = unwrapProxy(inst) or inst
			for _, target in ipairs({ realI, inst }) do
				local ok, b = pcall(getscriptbytecode, target)
				if ok and b and type(b) == "string" and #b > 4 then
					bc = b
					break
				end
			end
		end

		-- Bytecode hash dedup
		if bc ~= "" then
			local firstPath = global_seen_bc[bc]
			if firstPath then
				total_skip += 1
				updateStats()
				local thisPath = fullNameOf(inst)
				if thisPath ~= firstPath then
					table.insert(aliasLines, thisPath .. "  ->  " .. firstPath)
				end
				return
			end
			global_seen_bc[bc] = fullNameOf(inst)
		end

		seenPtr[inst] = true
		table.insert(queue, { inst = inst, bc = bc, tag = tag })
		scanTotal += 1
		setPhaseStatus("scan", scanTotal)
	end

	-- ── Metoda 1: getscripts() ──────────────────────────────────
	addLog("── [M1] getscripts()", Color3.fromRGB(150, 80, 255))
	if getscripts then
		local ok, scripts = pcall(getscripts)
		if ok and scripts then
			for _, s in ipairs(scripts) do
				if isScriptType(s) then tryAddToQueue(s, "getscripts") end
				checkYield()
			end
			addLog(string.format("     → %d scripts", #scripts), Color3.fromRGB(100, 200, 255))
		else
			addLog("     [ERR] " .. tostring(scripts), Color3.fromRGB(220, 80, 80))
		end
	else
		addLog("     [SKIP] unavailable", Color3.fromRGB(120, 120, 120))
	end

	-- ── Metoda 2: getloadedmodules() ───────────────────────────
	addLog("── [M2] getloadedmodules()", Color3.fromRGB(150, 80, 255))
	if getloadedmodules then
		local ok, mods = pcall(getloadedmodules)
		if ok and mods then
			for _, m in ipairs(mods) do
				if isScriptType(m) then tryAddToQueue(m, "loadedmodules") end
				checkYield()
			end
			addLog(string.format("     → %d modules", #mods), Color3.fromRGB(100, 200, 255))
		else
			addLog("     [SKIP] no modules", Color3.fromRGB(120, 120, 120))
		end
	else
		addLog("     [SKIP] unavailable", Color3.fromRGB(120, 120, 120))
	end

	-- ── Metoda 3: getinstances() ────────────────────────────────
	addLog("── [M3] getinstances()", Color3.fromRGB(150, 80, 255))
	if getinstances then
		local ok, insts = pcall(getinstances)
		if ok and insts then
			local cnt = 0
			for _, inst in ipairs(insts) do
				if isScriptType(inst) then tryAddToQueue(inst, "getinstances"); cnt += 1 end
				checkYield()
			end
			addLog(string.format("     → %d scripts", cnt), Color3.fromRGB(100, 200, 255))
		else
			addLog("     [SKIP] no instances", Color3.fromRGB(120, 120, 120))
		end
	else
		addLog("     [SKIP] unavailable", Color3.fromRGB(120, 120, 120))
	end

	-- ── Metoda 4: getnilinstances() ─────────────────────────────
	addLog("── [M4] getnilinstances()", Color3.fromRGB(150, 80, 255))
	if getnilinstances then
		local ok, nils = pcall(getnilinstances)
		if ok and nils then
			local cnt = 0
			for _, inst in ipairs(nils) do
				if isScriptType(inst) then tryAddToQueue(inst, "nilinstances"); cnt += 1 end
				checkYield()
			end
			addLog(string.format("     → %d nil-scripts", cnt), Color3.fromRGB(100, 200, 255))
		else
			addLog("     [SKIP] no nil-instances", Color3.fromRGB(120, 120, 120))
		end
	else
		addLog("     [SKIP] unavailable", Color3.fromRGB(120, 120, 120))
	end

	-- ── Method 5: GC - script instances ────────────────────────
	addLog("── [M5] getgc() – script instances", Color3.fromRGB(150, 80, 255))
	local okGC, gcData = pcall(getgc, true)
	sharedGC = okGC and gcData or nil
	if okGC and gcData then
		local cnt = 0
		for _, v in next, gcData do
			if type(v) == "userdata" then
				local ok2, tn = pcall(typeof, v)
				if ok2 and tn == "Instance" and isScriptType(v) then
					tryAddToQueue(v, "gc_instances")
					cnt += 1
				end
			end
			if _ % 5000 == 0 then checkYield() end
		end
		addLog(string.format("     → %d scripts in GC", cnt), Color3.fromRGB(100, 200, 255))
	else
		addLog("     [SKIP] getgc() unavailable", Color3.fromRGB(120, 120, 120))
	end

	-- ── Metoda 6: Game Tree Walk (wszystkie serwisy) ─────────────
	addLog("── [M6] Game Tree Walk", Color3.fromRGB(150, 80, 255))
	local serviceNames = {
		"Workspace","ReplicatedStorage","ReplicatedFirst","StarterGui",
		"StarterPack","StarterPlayer","Teams","SoundService","Chat",
		"Players","ServerScriptService","ServerStorage","Lighting","MaterialService",
	}
	for _, svcName in ipairs(serviceNames) do
		if Config.Services[svcName] == false then continue end
		local ok0, svc = pcall(function() return game:GetService(svcName) end)
		if not ok0 or not svc then continue end
		local okD, desc = pcall(function() return svc:GetDescendants() end)
		if okD and desc then
			local cnt = 0
			for _, inst in ipairs(desc) do
				if isScriptType(inst) then tryAddToQueue(inst, "gametree_" .. svcName); cnt += 1 end
				checkYield()
			end
			if cnt > 0 then
				addLog(string.format("     %s: %d scripts", svcName, cnt), Color3.fromRGB(100, 200, 255))
			end
		end
		checkYield()
	end

	addLog(string.format("✓ Scanning: %d unique scripts collected.", #queue), Color3.fromRGB(80, 255, 140))
	return queue
end

-- =================================================================
-- PHASE 2: DECOMPILE - Decompile all collected scripts
-- =================================================================
local function runDecompilePhase(queue)
	local results    = {}  -- { name, src, bc, gamePath, tag }
	local gcClosures = {}  -- { name, src }
	local seen_closure_key = {}
	local doneCount  = 0
	local medalFails = 0

	addLog(string.format("Decompiling %d scripts...", #queue), Color3.fromRGB(100, 200, 255))

	for _, item in ipairs(queue) do
		local inst     = item.inst
		local bc       = item.bc
		local realInst = unwrapProxy(inst) or inst

		-- Get name
		local name = ""
		pcall(function() name = realInst.Name end)
		if name == "" then pcall(function() name = inst.Name end) end
		local baseName = safeStr(name)

		-- Name filter
		if Config.NameFilterActive and Config.NameFilterText ~= "" then
			if not string.find(string.lower(baseName), string.lower(Config.NameFilterText), 1, true) then
				total_skip += 1; updateStats()
				doneCount += 1; setPhaseStatus("decompile", doneCount, #queue)
				continue
			end
		end

		-- Size filter: too big to decompile safely -> keep bytecode only
		if bc ~= "" and Config.MaxDecompileSize > 0 and #bc > Config.MaxDecompileSize then
			addLog("[BC-ONLY:SIZE] " .. baseName, Color3.fromRGB(240, 180, 100))
			local bigPath = getFullGamePath(realInst)
			if bigPath == "" then bigPath = getFullGamePath(inst) end
			table.insert(results, {
				name = baseName, src = nil, bc = bc, bcOnly = true,
				gamePath = bigPath, tag = item.tag, cls = classOf(realInst), rawName = name,
			})
			doneCount += 1; setPhaseStatus("decompile", doneCount, #queue)
			continue
		end

		-- Get 1:1 game path via GetFullName
		local gamePath = getFullGamePath(realInst)
		if gamePath == "" then gamePath = getFullGamePath(inst) end

		-- Decompile
		local src = nil
		local skipDueToFilter = false

		if Config.Decompile then
			if Config.UseMedalDecompiler and bc and #bc > 0 then
				local http_req = request or http_request or (syn and syn.request) or (http and http.request)
				if http_req then
					local ok, res = pcall(http_req, {
						Url = "http://127.0.0.1:3000/luau/decompile",
						Method = "POST",
						Body = bc,
						Headers = { ["Content-Type"] = "application/octet-stream" }
					})
					if ok and res and res.StatusCode == 200 and res.Body and not isUselessSrc(res.Body) then
						local dec = cleanSource(res.Body)
						if Config.KeywordFilterActive and Config.KeywordFilterText ~= "" then
							if not string.find(string.lower(dec), string.lower(Config.KeywordFilterText), 1, true) then
								skipDueToFilter = true
							end
						end
						if not skipDueToFilter then
							src = dec
						end
					else
						local errMsg = "Error"
						if not ok then errMsg = tostring(res)
						elseif res then errMsg = "HTTP " .. tostring(res.StatusCode) .. " " .. tostring(res.Body) end
						medalFails += 1
						if medalFails <= 3 then
							addLog("[MEDAL ERR] " .. baseName .. " -> " .. string.sub(errMsg, 1, 60), Color3.fromRGB(240, 100, 100))
							logToFile(OUT_DIR .. "/decompile_errors.log", "Name: " .. baseName .. "\nPath: " .. gamePath .. "\nMedal Error: " .. tostring(errMsg))
						end
						if medalFails == 3 then
							Config.UseMedalDecompiler = false
							addLog("[SYS] Medal unreachable 3x, switched off for this run (executor decompiler used)", Color3.fromRGB(255, 200, 80))
						end
					end
				else
					Config.UseMedalDecompiler = false
					addLog("[SYS] no HTTP function for Medal, switched off for this run", Color3.fromRGB(255, 200, 80))
				end
			end

			if not src and not skipDueToFilter and decompile then
				for _, target in ipairs({ realInst, inst }) do
					local ok, dec = pcall(decompile, target)
					if ok and dec and not isUselessSrc(dec) then
						dec = cleanSource(dec)
						-- Keyword filter
						if Config.KeywordFilterActive and Config.KeywordFilterText ~= "" then
							if not string.find(string.lower(dec), string.lower(Config.KeywordFilterText), 1, true) then
								skipDueToFilter = true
								break
							end
						end
						src = dec
						break
					elseif not ok and dec then
						logToFile(OUT_DIR .. "/decompile_errors.log",
							"Name: " .. baseName .. "\nPath: " .. gamePath .. "\nError: " .. tostring(dec))
					end
				end
			end
		end

		if skipDueToFilter then
			total_skip += 1; updateStats()
			doneCount += 1; setPhaseStatus("decompile", doneCount, #queue)
			continue
		end

		doneCount += 1
		setPhaseStatus("decompile", doneCount, #queue)

		if src or (Config.FallbackBytecode and bc ~= "") then
			table.insert(results, {
				name     = baseName,
				src      = src,
				bc       = bc,
				gamePath = gamePath,
				tag      = item.tag,
				cls      = classOf(realInst),
				rawName = name,
			})
		else
			total_skip += 1
			updateStats()
		end

		checkYield()
	end

	-- GC Closure scan (functions in memory, not bound to instances)
	addLog("── GC Closure scan...", Color3.fromRGB(150, 80, 255))
	local okGC, gcData = sharedGC ~= nil, sharedGC
	if okGC and gcData and decompile then
		local closureCnt = 0
		for _, v in next, gcData do
			if type(v) == "function" then
				local info
				pcall(function() info = debug.getinfo(v) end)
				if info and info.source and not string.match(info.source, "^%[C%]") then
					local key = info.source .. ":" .. tostring(info.linedefined or 0)
					if not seen_closure_key[key] then
						seen_closure_key[key] = true
						local ok2, dec = pcall(decompile, v)
						if ok2 and not isUselessSrc(dec) then
							dec = cleanSource(dec)
							local cleanSrc = safeStr(
								string.match(string.gsub(info.source, "^[@=]", ""), "[^/\\]+$") or info.source
							)
							table.insert(gcClosures, {
								name = cleanSrc .. "_L" .. tostring(info.linedefined or 0),
								src  = dec,
							})
							closureCnt += 1
						end
					end
				end
			end
			if _ % 5000 == 0 then checkYield() end
		end
		addLog(string.format("     → %d GC closures decompiled", closureCnt), Color3.fromRGB(100, 200, 255))
	end

	sharedGC = nil
	addLog(string.format("✓ Decompiling: %d results, %d GC closures.", #results, #gcClosures), Color3.fromRGB(80, 255, 140))
	return results, gcClosures
end

-- =================================================================
-- PHASE 3: WRITE - Save with exact 1:1 game tree structure
-- =================================================================
local function runWritePhase(results, gcClosures)
	local treeBase = OUT_DIR .. "/game_tree_1to1"
	ensureDir(OUT_DIR)
	ensureDir(treeBase)

	addLog(string.format("Writing %d scripts with 1:1 structure...", #results), Color3.fromRGB(100, 200, 255))

	for i, item in ipairs(results) do
		local name     = item.name
		local src      = item.src
		local bc       = item.bc
		local gamePath = item.gamePath  -- e.g. "ReplicatedStorage/WeaponSystem/ClientHandler"
		local tag      = item.tag or "unknown"

		-- Build folder from gamePath
		-- gamePath = "Service/Folder1/Folder2/ScriptName"
		-- → folder = treeBase/Service/Folder1/Folder2/
		-- → file   = treeBase/Service/Folder1/Folder2/ScriptName.lua
		local folderPath

		if gamePath ~= "" then
			local parts = string.split(gamePath, "/")
			if #parts > 1 then
				local folderParts = {}
				for j = 1, #parts - 1 do
					table.insert(folderParts, parts[j])
				end
				folderPath = treeBase .. "/" .. table.concat(folderParts, "/")
			else
				-- Single-level (script at root of service)
				folderPath = treeBase
			end
		else
			-- Unknown path → misc folder tagged by scan source
			folderPath = treeBase .. "/_misc/" .. safeStr(tag)
		end

		local writePath = folderPath .. "/" .. name

		if src then
			-- Threat scan
			if Config.ScanThreats then
				for _, threat in ipairs(ThreatKeywords) do
					if string.find(src, threat.pattern) then
						local preview = getLinePreview(src, threat.pattern)
						logToFile(OUT_DIR .. "/scanned_threats.log", string.format(
							"[POTENTIAL THREAT]\nScript: %s\nGamePath: %s\nType: %s\nPreview: %s",
							name .. ".lua", gamePath, threat.name, preview
						))
					end
				end
			end

			local finalPath = getUniquePath(writePath, ".lua")
			pcall(createFoldersForPath, finalPath)
			pcall(ensureDir, folderPath)
			local wok = pcall(writefile, finalPath, src)
			if wok then
				local shortPath = gamePath ~= "" and gamePath or ("_misc/" .. tag)
				if #shortPath > 55 then shortPath = "..." .. string.sub(shortPath, -52) end
				addLog("[✓] " .. name .. " → " .. shortPath, Color3.fromRGB(80, 255, 140))
				total_lua += 1
				updateStats()
			else
				addLog("[ERR:WRITE] " .. name, Color3.fromRGB(220, 80, 80))
				total_skip += 1
				updateStats()
			end
		elseif (Config.FallbackBytecode or item.bcOnly) and bc ~= "" then
			local finalPath = getUniquePath(writePath, ".luac")
			pcall(createFoldersForPath, finalPath)
			pcall(ensureDir, folderPath)
			local wok = pcall(writefile, finalPath, bc)
			if wok then
				addLog("[BC] " .. name .. ".luac", Color3.fromRGB(160, 100, 255))
				total_luac += 1
				updateStats()
			end
		else
			addLog("[FAIL] " .. name, Color3.fromRGB(220, 80, 80))
			total_skip += 1
			updateStats()
		end

		setPhaseStatus("write", i, #results)
		checkYield()
	end

	-- Write GC closures to separate folder
	if #gcClosures > 0 then
		local gcBase = OUT_DIR .. "/gc_closures"
		ensureDir(gcBase)
		addLog(string.format("Writing %d GC closures...", #gcClosures), Color3.fromRGB(150, 100, 220))
		for _, item in ipairs(gcClosures) do
			local basePath = gcBase .. "/" .. item.name
			local fname = getUniquePath(basePath, ".lua")
			local wok = pcall(writefile, fname, item.src)
			if wok then
				addLog("[GC] " .. (string.match(fname, "[^/]+$") or item.name), Color3.fromRGB(150, 100, 220))
				total_lua += 1
				updateStats()
			end
			checkYield()
		end
	end
end

-- EXTRA PHASE: ALL REMOTES + FULL INSTANCE TREE
-- =================================================================
local REMOTE_CLASSES = {
	RemoteEvent = true,
	UnreliableRemoteEvent = true,
	RemoteFunction = true,
}
local BINDABLE_CLASSES = {
	BindableEvent = true,
	BindableFunction = true,
}


-- CoreGui / CorePackages hold the executor's and Roblox's own UI: skip them in every listing
local coreRoots = nil
local function inCore(inst)
	if not coreRoots then
		coreRoots = {}
		for _, name in ipairs({ "CoreGui", "CorePackages" }) do
			local okS, svc = pcall(function() return game:GetService(name) end)
			if okS and svc then table.insert(coreRoots, svc) end
		end
	end
	for _, root in ipairs(coreRoots) do
		local okD, inside = pcall(function() return inst:IsDescendantOf(root) end)
		if okD and inside then return true end
	end
	return false
end
-- snapshot helpers: compare against the previous dump sitting in the same folder
local function readLines(path)
	local set = {}
	local ok, content = pcall(function()
		if isfile and isfile(path) then return readfile(path) end
		return nil
	end)
	if ok and type(content) == "string" then
		for line in string.gmatch(content .. "\n", "([^\n]*)\n") do
			if line ~= "" and string.sub(line, 1, 1) ~= "#" then
				set[line] = true
			end
		end
	end
	return set
end

local function writeDiff(path, label, oldSet, newLines)
	local newSet = {}
	for _, l in ipairs(newLines) do
		if l ~= "" and string.sub(l, 1, 1) ~= "#" then
			newSet[l] = true
		end
	end

	local out = {}
	if next(oldSet) == nil then
		table.insert(out, "no previous " .. label .. " snapshot, nothing to compare yet")
	else
		local added, removed = {}, {}
		for l in pairs(newSet) do
			if not oldSet[l] then table.insert(added, l) end
		end
		for l in pairs(oldSet) do
			if not newSet[l] then table.insert(removed, l) end
		end
		table.sort(added)
		table.sort(removed)
		table.insert(out, "ADDED " .. label .. ": " .. tostring(#added))
		for _, l in ipairs(added) do table.insert(out, "  + " .. l) end
		table.insert(out, "REMOVED " .. label .. ": " .. tostring(#removed))
		for _, l in ipairs(removed) do table.insert(out, "  - " .. l) end
	end

	ensureDir(OUT_DIR)
	pcall(writefile, path, table.concat(out, "\n"))
end

local function runRemotePhase()
	local oldRemotes = readLines(OUT_DIR .. "/remotes.txt")
	local seen, remotes, treeLines, bindables = {}, {}, {}, {}

	local function visit(inst, nilTag)
		if seen[inst] then return end
		seen[inst] = true
		if inCore(inst) then return end
		local ok, cn = pcall(function() return inst.ClassName end)
		if not ok or not cn then return end
		if not nilTag then
			local okP, par = pcall(function() return inst.Parent end)
			if okP and par == nil then nilTag = true end
		end
		local full = fullNameOf(inst)
		if nilTag then full = "[NIL] " .. full end
		if REMOTE_CLASSES[cn] then
			table.insert(remotes, { cn = cn, path = full })
		elseif BINDABLE_CLASSES[cn] then
			table.insert(bindables, { cn = cn, path = full })
		end
		if Config.DumpTree and #treeLines < Config.MaxTreeLines then
			table.insert(treeLines, cn .. "  " .. full)
		end
	end

	-- R1: everything under game
	addLog("── [R1] game:GetDescendants()", Color3.fromRGB(150, 80, 255))
	local okD, desc = pcall(function() return game:GetDescendants() end)
	if okD and desc then
		for _, inst in ipairs(desc) do
			visit(inst)
			checkYield()
		end
		addLog(string.format("     → %d instances", #desc), Color3.fromRGB(100, 200, 255))
	else
		addLog("     [ERR] GetDescendants failed", Color3.fromRGB(220, 80, 80))
	end

	-- R2: getinstances() catches what the tree walk can't reach
	if getinstances then
		local okI, insts = pcall(getinstances)
		if okI and insts then
			for _, inst in ipairs(insts) do
				visit(inst)
				checkYield()
			end
			addLog(string.format("── [R2] getinstances() → %d", #insts), Color3.fromRGB(100, 200, 255))
		end
	end

	-- R3: nil-parented instances (remotes held only by scripts)
	if getnilinstances then
		local okN, nils = pcall(getnilinstances)
		if okN and nils then
			for _, inst in ipairs(nils) do
				visit(inst, true)
				checkYield()
			end
			addLog(string.format("── [R3] getnilinstances() → %d", #nils), Color3.fromRGB(100, 200, 255))
		end
	end

	-- write remotes list
	table.sort(remotes, function(a, b) return a.path < b.path end)
	local rLines = { "# " .. #remotes .. " remotes", "" }
	for _, r in ipairs(remotes) do
		table.insert(rLines, string.format("[%s] %s", r.cn, r.path))
		addLog(string.format("[REMOTE] %s", r.path), Color3.fromRGB(80, 220, 240))
	end
	ensureDir(OUT_DIR)
	if Config.DumpRemotes then
		pcall(writefile, OUT_DIR .. "/remotes.txt", table.concat(rLines, "\n"))
		writeDiff(OUT_DIR .. "/changes_remotes.txt", "remotes", oldRemotes, rLines)
		table.sort(bindables, function(a, b) return a.path < b.path end)
		local bLines = { "# " .. tostring(#bindables) .. " bindables (not remotes)", "" }
		for _, b in ipairs(bindables) do
			table.insert(bLines, string.format("[%s] %s", b.cn, b.path))
		end
		pcall(writefile, OUT_DIR .. "/bindables.txt", table.concat(bLines, "\n"))
	end

	if Config.DumpTree then
		pcall(writefile, OUT_DIR .. "/game_tree.txt", table.concat(treeLines, "\n"))
	end

	return #remotes, #treeLines, remotes
end

-- INDEX PHASE: CALL SITES, VALUES/ATTRIBUTES, SUMMARY

local CALL_KINDS = { "FireServer", "InvokeServer", "FireClient", "FireAllClients", "InvokeClient" }
local RECV_KINDS = { "OnClientEvent", "OnClientInvoke" }

local function trimLine(s)
	s = string.gsub(s, "^%s+", "")
	s = string.gsub(s, "%s+$", "")
	return s
end

-- drop a trailing "-- comment" and trailing separators
local function cleanExpr(s)
	s = string.gsub(s, "%-%-.*$", "")
	s = string.gsub(s, "[;,]%s*$", "")
	return trimLine(s)
end

-- set -> sorted list, optionally capped with a "... (+N)" tail
local function setToList(set, cap)
	local list = {}
	for k in pairs(set) do
		table.insert(list, k)
	end
	table.sort(list)
	if cap and #list > cap then
		local short = {}
		for i = 1, cap do
			table.insert(short, list[i])
		end
		table.insert(short, "... (+" .. tostring(#list - cap) .. ")")
		return short
	end
	return list
end

local function joinSet(set, cap)
	local list = setToList(set, cap)
	if #list == 0 then return "-" end
	return table.concat(list, ", ")
end

-- best-effort: which instance name does an expression point at
local function nameFromExpr(expr)
	expr = cleanExpr(expr or "")
	local bracket = string.match(expr, "%[%s*[\"']([^\"']+)[\"']%s*%]$")
	if bracket then return bracket end
	local dotted = string.match(expr, "%.([%w_]+)$")
	if dotted then return dotted end
	local last = nil
	for nm in string.gmatch(expr, "WaitForChild%(%s*[\"']([^\"']+)") do
		last = nm
	end
	for nm in string.gmatch(expr, "FindFirstChild%(%s*[\"']([^\"']+)") do
		last = nm
	end
	if last then return last end
	return string.match(expr, "([%w_]+)$")
end

-- split "a, {b, c}, 'd'" on top-level commas (quotes and brackets respected)
local function splitArgs(text)
	local args, cur = {}, {}
	local depth, q = 0, nil
	local i, n = 1, #text
	while i <= n do
		local c = string.sub(text, i, i)
		if q then
			table.insert(cur, c)
			if c == "\\" and i < n then
				i += 1
				table.insert(cur, string.sub(text, i, i))
			elseif c == q then
				q = nil
			end
		elseif c == "\"" or c == "'" then
			q = c
			table.insert(cur, c)
		elseif c == "(" or c == "{" or c == "[" then
			depth = depth + 1
			table.insert(cur, c)
		elseif c == ")" or c == "}" or c == "]" then
			depth = depth - 1
			table.insert(cur, c)
		elseif c == "," and depth == 0 then
			table.insert(args, trimLine(table.concat(cur)))
			cur = {}
		else
			table.insert(cur, c)
		end
		i += 1
	end
	local last = trimLine(table.concat(cur))
	if last ~= "" then
		table.insert(args, last)
	end
	return args
end

-- text between the "(" at openPos and its matching ")", plus whether it closed
local function extractArgs(src, openPos)
	local depth, q = 0, nil
	local i = openPos
	local limit = math.min(#src, openPos + 1500)
	while i <= limit do
		local c = string.sub(src, i, i)
		if q then
			if c == "\\" then
				i += 1
			elseif c == q then
				q = nil
			end
		elseif c == "\"" or c == "'" then
			q = c
		elseif c == "(" then
			depth = depth + 1
		elseif c == ")" then
			depth = depth - 1
			if depth == 0 then
				return string.sub(src, openPos + 1, i - 1), true
			end
		end
		i += 1
	end
	return string.sub(src, openPos + 1, limit), false
end

local function buildLineIndex(src)
	local starts = { 1 }
	local pos = 1
	while true do
		local nl = string.find(src, "\n", pos, true)
		if not nl then break end
		table.insert(starts, nl + 1)
		pos = nl + 1
	end
	return starts
end

-- 1-based line number that contains byte position pos
local function lineOf(starts, pos)
	local lo, hi = 1, #starts
	while lo < hi do
		local mid = math.floor((lo + hi + 1) / 2)
		if starts[mid] <= pos then
			lo = mid
		else
			hi = mid - 1
		end
	end
	return lo
end

-- chain text (Remotes.Foo, Remote, Remotes["X"], WaitForChild("X")) -> remote name
local function resolveRemote(src, chain, cache)
	if cache[chain] then return cache[chain] end
	local name = nil
	if string.match(chain, "^[%w_]+$") then
		-- plain variable: follow its local assignment once
		local rhs = string.match(src, "local%s+" .. chain .. "%s*=%s*([^\n]+)")
		if rhs then name = nameFromExpr(rhs) end
	else
		name = nameFromExpr(chain)
	end
	name = name or chain
	cache[chain] = name
	return name
end

-- receiver expression sitting right before ":Method(" or ".Event" (pre = text before it)
local function chainAt(pre)
	return string.match(pre, "([%w_]+%b())$")
		or string.match(pre, "([%w_%.]*%[%s*[\"'][^\"'\n]*[\"']%s*%])$")
		or string.match(pre, "([%w_%.%[%]\"']+)$")
		or "?"
end

-- every remote call (out) and handler (in) inside one decompiled source
local function scanRemoteSites(src, where, info, ctx)
	local starts = nil
	local cache = {}

	local function addSite(kind, rname, dir, pos, detail)
		local u = ctx.usage[rname]
		if not u then
			u = { out = {}, ["in"] = {} }
			ctx.usage[rname] = u
		end
		u[dir][where] = true
		if dir == "out" then
			ctx.byOut[rname] = (ctx.byOut[rname] or 0) + 1
			info.outRemotes[rname] = true
		else
			ctx.byIn[rname] = (ctx.byIn[rname] or 0) + 1
			info.inRemotes[rname] = true
		end
		ctx.totalCalls = ctx.totalCalls + 1

		if ctx.callLinesCount < Config.MaxCallSites then
			ctx.callLinesCount = ctx.callLinesCount + 1
			if not starts then starts = buildLineIndex(src) end
			local line = lineOf(starts, pos)
			table.insert(ctx.callLines, string.format("[%s] %s  @ %s : line %d", kind, rname, where, line))
			table.insert(ctx.callLines, "    " .. detail)
		end
	end

	for _, kind in ipairs(CALL_KINDS) do
		local needle = ":" .. kind .. "("
		local pos = 1
		while true do
			local s = string.find(src, needle, pos, true)
			if not s then break end
			pos = s + #needle
			local pre = string.sub(src, math.max(1, s - 160), s - 1)
			local chain = chainAt(pre)
			local rname = resolveRemote(src, chain, cache)
			local argText, complete = extractArgs(src, pos - 1)
			local flat = string.gsub(argText, "%s+", " ")
			local parts = {}
			for ai, a in ipairs(splitArgs(flat)) do
				if ai > 12 then break end
				table.insert(parts, "[" .. tostring(ai) .. "] " .. string.sub(a, 1, 160))
			end
			local detail = "args: " .. (#parts > 0 and table.concat(parts, "  ") or "(none)")
			if not complete then detail = detail .. "  ...(cut)" end
			addSite(kind, rname, "out", s, detail)
		end
	end

	for _, kind in ipairs(RECV_KINDS) do
		local needle = "." .. kind
		local pos = 1
		while true do
			local s, e = string.find(src, needle, pos, true)
			if not s then break end
			pos = e + 1
			local pre = string.sub(src, math.max(1, s - 160), s - 1)
			local chain = chainAt(pre)
			local rname = resolveRemote(src, chain, cache)
			local ahead = string.sub(src, e + 1, e + 240)
			local params = string.match(ahead, "^%s*:%s*Connect%s*%(%s*function%s*%(([^%)]*)%)")
				or string.match(ahead, "^%s*=%s*function%s*%(([^%)]*)%)")
				or "?"
			addSite(kind, rname, "in", s, "params: (" .. params .. ")")
		end
	end
end

-- values, attributes, object props and tags across the whole game
local function dumpValues()
	local VALUE_CLASSES = {
		BoolValue = true, IntValue = true, NumberValue = true, StringValue = true,
		ObjectValue = true, Vector3Value = true, CFrameValue = true,
		Color3Value = true, BrickColorValue = true,
	}
	local OBJECT_PROPS = {
		ProximityPrompt = { "ActionText", "ObjectText", "KeyboardKeyCode", "HoldDuration", "MaxActivationDistance", "Enabled", "RequiresLineOfSight" },
		ClickDetector   = { "MaxActivationDistance" },
		Tool            = { "RequiresHandle", "CanBeDropped", "Enabled", "ToolTip" },
		Animation       = { "AnimationId" },
		Sound           = { "SoundId", "Volume", "Looped" },
		Team            = { "TeamColor", "AutoAssignable" },
		SpawnLocation   = { "Enabled", "Neutral" },
		Humanoid        = { "WalkSpeed", "JumpPower", "JumpHeight", "MaxHealth", "HipHeight" },
	}

	local valLines, objLines, tagTable = {}, {}, {}
	local count = 0

	local okD, desc = pcall(function() return game:GetDescendants() end)
	if okD and desc then
		for _, inst in ipairs(desc) do
			if not inCore(inst) then
				local cn = classOf(inst)

				if VALUE_CLASSES[cn] then
					local okV, v = pcall(function() return inst.Value end)
					local vs = "?"
					if okV then
						if typeof(v) == "Instance" then vs = fullNameOf(v) else vs = tostring(v) end
					end
					table.insert(valLines, string.format("[%s] %s = %s", cn, fullNameOf(inst), vs))
					count += 1
				end

				local props = OBJECT_PROPS[cn]
				if props then
					local parts = {}
					for _, p in ipairs(props) do
						local okP, pv = pcall(function() return inst[p] end)
						table.insert(parts, p .. "=" .. (okP and tostring(pv) or "?"))
					end
					table.insert(objLines, string.format("[%s] %s  |  %s", cn, fullNameOf(inst), table.concat(parts, ", ")))
				end

				local okA, attrs = pcall(function() return inst:GetAttributes() end)
				if okA and attrs then
					for k, v in pairs(attrs) do
						table.insert(valLines, string.format("[ATTR] %s @ %s = %s (%s)", tostring(k), fullNameOf(inst), tostring(v), typeof(v)))
						count += 1
					end
				end

				local okT, tags = pcall(function() return inst:GetTags() end)
				if okT and tags then
					for _, t in ipairs(tags) do
						local e = tagTable[t]
						if not e then
							e = { count = 0, example = fullNameOf(inst) }
							tagTable[t] = e
						end
						e.count = e.count + 1
					end
				end
			end
			checkYield()
		end
	end

	table.sort(valLines)
	table.sort(objLines)
	ensureDir(OUT_DIR)
	pcall(writefile, OUT_DIR .. "/values.txt", table.concat(valLines, "\n"))
	pcall(writefile, OUT_DIR .. "/objects.txt", table.concat(objLines, "\n"))
	return count, tagTable
end

-- live state of the runtime hooks and listeners, shown in meta.txt
local RuntimeState = { outgoingHooks = 0, incomingListeners = 0 }
-- who we are, where we are, on which build

	add("Character", function() return char.Name end)
	add("Humanoid", function() return hum.Health .. "/" .. hum.MaxHealth end)
	add("WalkSpeed", function() return hum.WalkSpeed end)
	add("JumpPower", function() return hum.JumpPower end)
	add("JumpHeight", function() return hum.JumpHeight end)

	ensureDir(OUT_DIR)
	pcall(writefile, OUT_DIR .. "/meta.txt", table.concat(lines, "\n"))
	return lines
end

local function runIndexPhase(results, remoteList)
	local prevScripts = readLines(OUT_DIR .. "/scripts.txt")

	local ctx = {
		callLines = {}, callLinesCount = 0, totalCalls = 0,
		byOut = {}, byIn = {}, usage = {},
	}
	local allAttrs, allTags = {}, {}
	local scriptInfos = {}
	local totalReqs = 0

	for _, item in ipairs(results) do
		local where = (item.gamePath ~= "" and item.gamePath) or ("_misc/" .. tostring(item.tag))
		local info = {
			path = where, cls = item.cls or "?", rawName = item.rawName, lines = 0,
			outRemotes = {}, inRemotes = {}, reqs = {}, funcs = {},
			attrs = {}, tags = {}, refs = {}, services = {},
		}

		if item.src then
			local okSite, errSite = pcall(scanRemoteSites, item.src, where, info, ctx)
			if not okSite then
				addLog("[ERR] sites " .. where .. ": " .. tostring(errSite), Color3.fromRGB(220, 80, 80))
			end

			local n = 0
			for line in string.gmatch(item.src .. "\n", "([^\n]*)\n") do
				n += 1
				for a in string.gmatch(line, "Attribute%w*%(%s*[\"']([^\"']+)") do
					allAttrs[a] = true
					info.attrs[a] = true
				end
				for t in string.gmatch(line, "Tag%w*%(%s*[\"']([^\"']+)") do
					allTags[t] = true
					info.tags[t] = true
				end
				for r in string.gmatch(line, "WaitForChild%(%s*[\"']([^\"']+)") do
					info.refs[r] = true
				end
				for r in string.gmatch(line, "FindFirstChild%(%s*[\"']([^\"']+)") do
					info.refs[r] = true
				end
				for sv in string.gmatch(line, "GetService%(%s*[\"']([^\"']+)") do
					info.services[sv] = true
				end
				local fname = string.match(line, "function%s+([%w_%.:]+)%s*%(")
					or string.match(line, "([%w_%.:]+)%s*=%s*function")
				if fname then
					info.funcs[fname] = true
				end
				for req in string.gmatch(line, "require%s*%((.-)%)") do
					info.reqs[cleanExpr(req)] = true
					totalReqs += 1
				end
				if n % 500 == 0 then checkYield() end
			end
			info.lines = n
		end

		table.insert(scriptInfos, info)
		checkYield()
	end

	local okDV, valueCount, tagTable = pcall(dumpValues)
	if not okDV then
		addLog("[ERR] values: " .. tostring(valueCount), Color3.fromRGB(220, 80, 80))
		valueCount, tagTable = 0, {}
	end
	local okM, metaLines = pcall(writeMeta)
	if not okM then metaLines = {} end

	-- require(): map each expression to a dumped script when the name is unique
	local nameToPaths = {}
	for _, s in ipairs(scriptInfos) do
		if s.rawName then
			nameToPaths[s.rawName] = nameToPaths[s.rawName] or {}
			table.insert(nameToPaths[s.rawName], s.path)
		end
	end
	local requiredBy = {}
	for _, s in ipairs(scriptInfos) do
		s.reqResolved = {}
		for raw in pairs(s.reqs) do
			local nm = nameFromExpr(raw)
			local targets = nil
			if nm then targets = nameToPaths[nm] end
			if targets and #targets == 1 then
				s.reqResolved[raw] = targets[1]
				requiredBy[targets[1]] = requiredBy[targets[1]] or {}
				requiredBy[targets[1]][s.path] = true
			elseif targets then
				s.reqResolved[raw] = "ambiguous (" .. tostring(#targets) .. " matches)"
			else
				s.reqResolved[raw] = "not in dump"
			end
		end
	end

	local svcLines = {}
	for _, svc in ipairs(game:GetChildren()) do
		local okS, d = pcall(function() return #svc:GetDescendants() end)
		table.insert(svcLines, string.format("  %-24s %s", svc.Name, okS and tostring(d) or "?"))
		checkYield()
	end
	table.sort(svcLines)

	local remLines, known = {}, {}
	for _, p in ipairs(remoteList or {}) do
		local short = string.match(p.path, "([^%.]+)$") or p.path
		known[short] = true
		table.insert(remLines, string.format("  %-28s out:%-4d in:%-4d [%s] %s",
			short, ctx.byOut[short] or 0, ctx.byIn[short] or 0, p.cn, p.path))
	end
	for name, cnt in pairs(ctx.byOut) do
		if not known[name] then
			known[name] = true
			table.insert(remLines, string.format("  %-28s out:%-4d in:%-4d [not in remote list]", name, cnt, ctx.byIn[name] or 0))
		end
	end
	for name, cnt in pairs(ctx.byIn) do
		if not known[name] then
			known[name] = true
			table.insert(remLines, string.format("  %-28s out:%-4d in:%-4d [not in remote list]", name, ctx.byOut[name] or 0, cnt))
		end
	end
	table.sort(remLines)

	local shortCount = {}
	for _, p in ipairs(remoteList or {}) do
		local sn = string.match(p.path, "([^%.]+)$") or p.path
		shortCount[sn] = (shortCount[sn] or 0) + 1
	end
	local usageLines = {}
	for _, name in ipairs(setToList(ctx.usage)) do
		local u = ctx.usage[name]
		local copies = shortCount[name] or 0
		table.insert(usageLines, copies > 1 and (name .. "  [" .. tostring(copies) .. " remotes share this name]") or name)
		table.insert(usageLines, "    fired from: " .. joinSet(u.out, 30))
		table.insert(usageLines, "    handled by: " .. joinSet(u["in"], 30))
	end

	local tagLines, tagKeys = {}, 0
	for t, e in pairs(tagTable) do
		tagKeys += 1
		table.insert(tagLines, string.format("  %-28s x%-6d e.g. %s", t, e.count, e.example))
	end
	table.sort(tagLines)

	table.sort(scriptInfos, function(a, b) return a.path < b.path end)
	local scLines, scPaths = {}, {}
	for _, s in ipairs(scriptInfos) do
		table.insert(scPaths, string.format("[%s] %s", s.cls, s.path))
		table.insert(scLines, string.format("[%s] %s  (%d lines)", s.cls, s.path, s.lines))

		local rset = {}
		for k in pairs(s.outRemotes) do rset[k .. " [out]"] = true end
		for k in pairs(s.inRemotes) do rset[k .. " [in]"] = true end

		local rows = {
			{ "remotes:",  rset,                     40 },
			{ "funcs:",    s.funcs,                  40 },
			{ "attrs:",    s.attrs,                  40 },
			{ "tags:",     s.tags,                   40 },
			{ "services:", s.services,               20 },
			{ "refs:",     s.refs,                   60 },
			{ "usedby:",   requiredBy[s.path] or {}, 40 },
		}
		for _, row in ipairs(rows) do
			local list = setToList(row[2], row[3])
			if #list > 0 then
				table.insert(scLines, string.format("    %-10s %s", row[1], table.concat(list, ", ")))
			end
		end

		local reqRows = {}
		for raw, target in pairs(s.reqResolved) do
			table.insert(reqRows, raw .. " -> " .. target)
		end
		table.sort(reqRows)
		for ri, r in ipairs(reqRows) do
			if ri > 40 then break end
			table.insert(scLines, "    requires:  " .. r)
		end
	end

	ensureDir(OUT_DIR)
	pcall(writefile, OUT_DIR .. "/scripts.txt", table.concat(scPaths, "\n"))
	writeDiff(OUT_DIR .. "/changes_scripts.txt", "scripts", prevScripts, scPaths)
	pcall(writefile, OUT_DIR .. "/call_sites.txt", table.concat(ctx.callLines, "\n"))
	pcall(writefile, OUT_DIR .. "/remote_usage.txt", table.concat(usageLines, "\n"))
	pcall(writefile, OUT_DIR .. "/tags.txt", table.concat(tagLines, "\n"))
	if #aliasLines > 0 then
		pcall(writefile, OUT_DIR .. "/aliases.txt", table.concat(aliasLines, "\n"))
	end

	local summary = {
		"SUPER DUMPER INDEX",
		"==================",
		"files: summary.txt meta.txt remotes.txt remote_usage.txt call_sites.txt values.txt objects.txt tags.txt scripts.txt aliases.txt changes_*.txt captured_args.txt bindables.txt",
		"note: Script (server) bodies are not readable from the client, so server handlers only show through the client's calls.",
		"",
	}
	for _, l in ipairs(metaLines) do table.insert(summary, l) end
	table.insert(summary, "")
	table.insert(summary, string.format(
		"Scripts: %d | Remotes: %d | Values/Attrs: %d | Tags: %d | Call sites: %d | Requires: %d | Aliases: %d",
		#results, #(remoteList or {}), valueCount, tagKeys, ctx.totalCalls, totalReqs, #aliasLines))
	table.insert(summary, "")
	table.insert(summary, "SERVICES (descendant count)")
	table.insert(summary, table.concat(svcLines, "\n"))
	table.insert(summary, "")
	table.insert(summary, "REMOTES (out = sent from code, in = handler on this side)")
	table.insert(summary, table.concat(remLines, "\n"))
	table.insert(summary, "")
	table.insert(summary, "REMOTE USAGE (which scripts fire / handle each remote)")
	table.insert(summary, table.concat(usageLines, "\n"))
	table.insert(summary, "")
	table.insert(summary, "ATTRIBUTES used in code: " .. joinSet(allAttrs, 100))
	table.insert(summary, "TAGS used in code: " .. joinSet(allTags, 100))
	table.insert(summary, "")
	table.insert(summary, "SCRIPTS")
	table.insert(summary, table.concat(scLines, "\n"))
	pcall(writefile, OUT_DIR .. "/summary.txt", table.concat(summary, "\n"))

	return ctx.totalCalls, totalReqs, valueCount
end

local function executeDump()
	total_lua   = 0
	total_luac  = 0
	total_skip  = 0
	global_seen_bc = {}
	aliasLines = {}
	sharedGC = nil
	table.clear(allLogLabels)

	for _, child in ipairs(LogScroll:GetChildren()) do
		if child:IsA("TextLabel") then child:Destroy() end
	end

	updateStats()
	resetPhaseUI()
	lastYield = os.clock()

	local startTime = os.clock()
	addLog("=== SUPER DUMPER V5 – 3-PHASE SYSTEM ===", Color3.fromRGB(150, 80, 255))
	addLog(string.format(
		"[SYS] decompile=%s | getscriptbytecode=%s | getscripts=%s | getinstances=%s",
		tostring(decompile ~= nil),
		tostring(getscriptbytecode ~= nil),
		tostring(getscripts ~= nil),
		tostring(getinstances ~= nil)
	), Color3.fromRGB(180, 180, 220))

	ensureDir(OUT_DIR)

	-- ══════════════════════════════════════════════════════════════
	-- ETAP 1: SKANOWANIE
	-- ══════════════════════════════════════════════════════════════
	setProgress("PHASE 1/3: Scanning all sources...")
	addLog("", Color3.fromRGB(40, 40, 40))
	addLog("╔══════════════════════════════════════╗", Color3.fromRGB(150, 80, 255))
	addLog("║  ⚡ PHASE 1: SCANNING (6 methods)     ║", Color3.fromRGB(150, 80, 255))
	addLog("╚══════════════════════════════════════╝", Color3.fromRGB(150, 80, 255))
	markPhaseActive("scan")
	switchTab("Logs")

	local queue = {}
	local okS, errS = pcall(runScanPhase, queue)
	if not okS then addLog("[ERR] scan: " .. tostring(errS), Color3.fromRGB(220, 80, 80)) end

	markPhaseDone("scan", #queue)
	addLog("", Color3.fromRGB(40, 40, 40))

	-- ══════════════════════════════════════════════════════════════
	-- ETAP 2: DEKOMPILACJA
	-- ══════════════════════════════════════════════════════════════
	setProgress("PHASE 2/3: Decompiling " .. tostring(#queue) .. " scripts...")
	addLog("╔══════════════════════════════════════╗", Color3.fromRGB(100, 180, 255))
	addLog("║  🧪 PHASE 2: DECOMPILING              ║", Color3.fromRGB(100, 180, 255))
	addLog("╚══════════════════════════════════════╝", Color3.fromRGB(100, 180, 255))
	markPhaseActive("decompile")

	local okD, results, gcClosures = pcall(runDecompilePhase, queue)
	if not okD then
		addLog("[ERR] decompile: " .. tostring(results), Color3.fromRGB(220, 80, 80))
		results, gcClosures = {}, {}
	end

	markPhaseDone("decompile", #results)
	addLog("", Color3.fromRGB(40, 40, 40))

	-- ══════════════════════════════════════════════════════════════
	-- ETAP 3: ZAPIS 1:1 STRUKTURA GRY
	-- ══════════════════════════════════════════════════════════════
	setProgress("PHASE 3/3: Recreating 1:1 structure...")
	addLog("╔══════════════════════════════════════╗", Color3.fromRGB(80, 255, 140))
	addLog("║  📁 PHASE 3: WRITING 1:1 STRUCTURE    ║", Color3.fromRGB(80, 255, 140))
	addLog("╚══════════════════════════════════════╝", Color3.fromRGB(80, 255, 140))
	markPhaseActive("write")

	local okW, errW = pcall(runWritePhase, results, gcClosures)
	if not okW then addLog("[ERR] write: " .. tostring(errW), Color3.fromRGB(220, 80, 80)) end

	markPhaseDone("write", total_lua)

	-- ── EXTRA: REMOTES + FULL TREE ─────────────────────────────
	setProgress("EXTRA: Remotes + full tree...")
	addLog("╔══════════════════════════════════════╗", Color3.fromRGB(255, 200, 80))
	addLog("║  🌐 EXTRA: REMOTES + FULL TREE        ║", Color3.fromRGB(255, 200, 80))
	addLog("╚══════════════════════════════════════╝", Color3.fromRGB(255, 200, 80))
	local okR, remoteCount, treeCount, remoteList = pcall(runRemotePhase)
	if not okR then
		addLog("[ERR] remote phase: " .. tostring(remoteCount), Color3.fromRGB(220, 80, 80))
		remoteCount, treeCount, remoteList = 0, 0, {}
	end

	-- ── EXTRA: INDEX (call sites, values, summary) ─────────────
	setProgress("INDEX: call sites + values + summary...")
	addLog("╔══════════════════════════════════════╗", Color3.fromRGB(255, 160, 220))
	addLog("║  🗂 INDEX: CALL SITES + SUMMARY       ║", Color3.fromRGB(255, 160, 220))
	addLog("╚══════════════════════════════════════╝", Color3.fromRGB(255, 160, 220))
	local okI, callCount, reqCount, valCount = pcall(runIndexPhase, results, remoteList or {})
	if not okI then
		addLog("[ERR] index phase: " .. tostring(callCount), Color3.fromRGB(220, 80, 80))
		callCount, reqCount, valCount = 0, 0, 0
	end

	-- ── PODSUMOWANIE ──────────────────────────────────────────────
	local elapsed = string.format("%.2fs", os.clock() - startTime)
	TimeElapsedText.Text = elapsed
	setProgress("All 3 phases completed!")
	addLog("", Color3.fromRGB(40, 40, 40))
	addLog("════════════════════════════════════════", Color3.fromRGB(150, 80, 255))
	addLog(string.format("✓ DUMP COMPLETED IN %s!", elapsed), Color3.fromRGB(80, 255, 140))
	addLog(string.format("  Saved: %d .lua  |  %d .luac  |  %d skipped", total_lua, total_luac, total_skip), Color3.fromRGB(200, 200, 200))
	addLog("  Struktura: workspace/" .. OUT_DIR .. "/game_tree_1to1/", Color3.fromRGB(180, 180, 255))
	addLog("  GC:        workspace/" .. OUT_DIR .. "/gc_closures/",    Color3.fromRGB(180, 180, 255))
	addLog(string.format("  Remotes: %d  |  Tree: %d instances  ->  workspace/%s/", remoteCount, treeCount, OUT_DIR), Color3.fromRGB(180, 255, 180))
	addLog(string.format("  Calls: %d  |  Requires: %d  |  Values/Attrs: %d  ->  workspace/%s/summary.txt", callCount, reqCount, valCount, OUT_DIR), Color3.fromRGB(255, 190, 230))

end

-- =================================================================
-- RUNTIME CAPTURE: outgoing hook + incoming OnClientEvent listeners
-- =================================================================
do
	local CAPTURE_FILE = OUT_DIR .. "/captured_args.txt"
	local CAPTURE_LIMIT = 50000
	local spyInitialized = false
	local incomingConns = {}
	local capturedCount = 0

	local function serializeValue(v, depth)
		depth = depth or 0
		local t = typeof(v)
		if t == "string" then
			return "\"" .. v .. "\""
		elseif t == "number" or t == "boolean" or t == "nil" then
			return tostring(v)
		elseif t == "Instance" then
			return "<" .. fullNameOf(v) .. ">"
		elseif t == "table" then
			if depth >= 3 then return "{...}" end
			local parts, n = {}, 0
			for k, val in pairs(v) do
				n += 1
				if n > 20 then
					table.insert(parts, "...")
					break
				end
				table.insert(parts, tostring(k) .. " = " .. serializeValue(val, depth + 1))
			end
			return "{" .. table.concat(parts, ", ") .. "}"
		end
		return t .. "(" .. tostring(v) .. ")"
	end

	local function formatArgs(...)
		local n = select("#", ...)
		if n == 0 then return "(no args)" end
		local args = { ... }
		local parts = {}
		for i = 1, n do
			table.insert(parts, "[" .. tostring(i) .. "] " .. serializeValue(args[i]))
		end
		return table.concat(parts, "  ")
	end

	local function captureWrite(text)
		if capturedCount >= CAPTURE_LIMIT then return end
		capturedCount += 1
		logToFile(CAPTURE_FILE, text)
		if capturedCount % 50 == 0 then
			addLog("[CAPTURE] " .. tostring(capturedCount) .. " events logged", Color3.fromRGB(80, 220, 240))
		end
	end

	local function hookOutgoing(className, methodName, label)
		local ok, installed = pcall(function()
			if not hookfunction then return false end
			local probe = Instance.new(className)
			local original = probe[methodName]
			probe:Destroy()
			if type(original) ~= "function" then return false end
			local old
			old = hookfunction(original, function(self, ...)
				if Config.NetworkSpyActive and capturedCount < CAPTURE_LIMIT then
					-- capture must never break the game's own call: anything that fails in here is swallowed
					pcall(function(...)
						local selfPath = fullNameOf(self)
						local callerPath = "?"
						if getcallingscript then
							local okC, cs = pcall(getcallingscript)
							if okC and cs then callerPath = fullNameOf(cs) end
						end
						local argText = formatArgs(...)
						task.spawn(function()
							captureWrite(string.format("[%s] %s  caller=%s\n    %s", label, selfPath, callerPath, argText))
						end)
					end, ...)
				end
				return old(self, ...)
			end)
			return true
		end)
		return ok and installed == true
	end

	function initializeNetworkSpy()
		if spyInitialized then return end
		spyInitialized = true
		local a = hookOutgoing("RemoteEvent", "FireServer", "FIRE")
		local b = hookOutgoing("UnreliableRemoteEvent", "FireServer", "FIRE-U")
		local c = hookOutgoing("RemoteFunction", "InvokeServer", "INVOKE")
		local count = (a and 1 or 0) + (b and 1 or 0) + (c and 1 or 0)
		RuntimeState.outgoingHooks = count
		if count > 0 then
			addLog(string.format("[SYS] Outgoing capture: %d/3 hooks -> %s", count, CAPTURE_FILE), Color3.fromRGB(80, 250, 200))
		else
			addLog("[SYS] Outgoing capture failed (no hookfunction in executor).", Color3.fromRGB(250, 100, 100))
		end
	end

	function setIncomingCapture(on)
		for _, conn in ipairs(incomingConns) do
			pcall(function() conn:Disconnect() end)
		end
		incomingConns = {}
		RuntimeState.incomingListeners = 0
		if not on then
			addLog("[SYS] Incoming capture OFF", Color3.fromRGB(180, 180, 180))
			return
		end
		local count = 0
		local okD, desc = pcall(function() return game:GetDescendants() end)
		if okD and desc then
			for _, inst in ipairs(desc) do
				local cn = classOf(inst)
				if cn == "RemoteEvent" or cn == "UnreliableRemoteEvent" then
					local path = fullNameOf(inst)
					local okC, conn = pcall(function()
						return inst.OnClientEvent:Connect(function(...)
							captureWrite("[IN] " .. path .. "\n    " .. formatArgs(...))
						end)
					end)
					if okC and conn then
						table.insert(incomingConns, conn)
						count += 1
					end
				end
			end
		end
		RuntimeState.incomingListeners = count
		addLog(string.format("[SYS] Incoming capture ON: %d remote events listening (re-toggle after new remotes appear)", count), Color3.fromRGB(80, 250, 200))
	end
end
